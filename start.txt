# Starts API + Web in two new terminal windows.
# Run from this folder (company-shehwa) or via the cmd below.

$root = $PWD.Path
if (-not (Test-Path (Join-Path $root "apps\api"))) {
  $root = Join-Path $root "company-shehwa"
}

$api = Join-Path $root "apps\api"
$web = Join-Path $root "apps\web"

if (-not (Test-Path $api)) { throw "API folder not found: $api" }
if (-not (Test-Path $web)) { throw "Web folder not found: $web" }

Start-Process cmd -ArgumentList "/k", "cd /d `"$api`" && npm run start:dev"
Start-Process cmd -ArgumentList "/k", "cd /d `"$web`" && npm run dev"

Write-Host "Started API (Nest) and Web (Vite) in new windows."
